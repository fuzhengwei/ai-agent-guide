第12章 RAG：检索增强生成

第三篇：Agent 的知识 — 让 Agent 拥有"外挂知识库"——从向量检索到生产级 RAG

12.1 为什么 Agent 需要 RAG？

大模型有两个硬伤：知识截止日期（训练数据有时效性）和领域知识不足（通用模型不懂你的业务）。微调可以解决一部分，但成本高、周期长、更新慢。RAG（Retrieval-Augmented Generation，检索增强生成）提供了更灵活的方案：先检索，再生成。

💡 RAG 的核心思想

不是把知识"塞进"模型参数，而是在生成回答前，先从外部知识库检索相关文档，把检索到的内容作为上下文喂给 LLM。模型只需要"阅读理解"，不需要"死记硬背"。

🔴 纯 LLM（无 RAG）
  • 知识截止到训练日期，不知道最新信息
  • 企业内部文档一概不知
  • 容易产生幻觉（编造事实）
  • 无法引用信息来源
  • 更新知识需要重新训练或微调
🟢 LLM + RAG
  • 知识实时更新，随时新增文档
  • 接入企业内部知识库（Wiki/文档/工单）
  • 基于检索到的真实内容回答，减少幻觉
  • 可以引用来源文档，可追溯
  • 无需重新训练，成本极低

RAG vs 微调：什么时候用什么？

对比维度 RAG 微调
知识更新实时，加文档即可需要重新微调
适合场景事实性问答、文档检索风格/格式/推理模式定制
成本低（向量数据库 + 检索）高（GPU 训练）
延迟多一次检索（+200-500ms）无额外延迟
幻觉控制✅ 强（基于检索内容）⚠️ 弱（仍可能编造）
最佳实践RAG + 微调结合：微调改善风格，RAG 提供知识

12.2 RAG 的完整流程

RAG 不是简单的"搜索 + 拼接"，一个生产级 RAG 系统包含五个阶段：

func splitText(text string, chunkSize, chunkOverlap int, separators []string) []string {
	var chunks []string
	for len(text) > chunkSize {
		// 在 chunkSize 范围内找最佳分割点
		end := chunkSize
		for _, sep := range separators {
			idx := strings.LastIndex(text[:end], sep)
			 if idx > 0 {
				end = idx + len(sep)
				break
			}
		}
		chunks = append(chunks, text[:end])
		// 保留 overlap 重叠部分
		start := end - chunkOverlap
		if start < 0 {
			start = 0
		}
		text = text[start:]
	}
	if len(text) > 0 {
		chunks = append(chunks, text)
	}
	return chunks
}

import java.util.*;

public class TextSplitter {

    // 递归字符分割器：按优先级依次尝试分隔符
    public static List<String> splitText(String text, int chunkSize, int chunkOverlap, List<String> separators) {
        List<String> chunks = new ArrayList<>();
        while (text.length() > chunkSize) {
            int end = chunkSize;
            for (String sep : separators) {
                int idx = text.substring(0, end).lastIndexOf(sep);
                if (idx > 0) {
                    end = idx + sep.length();
                    break;
                }
            }
            chunks.add(text.substring(0, end));
            int start = Math.max(0, end - chunkOverlap);
            text = text.substring(start);
        }
        if (!text.isEmpty()) {
            chunks.add(text);
        }
        return chunks;
    }

type EmbeddingRequest struct {

type EmbeddingResponse struct {
	Data []struct {
		Embedding []float64 `json:"embedding"`
	} `json:"data"`
}

func embed(texts []string) ([][]float64, error) {
	reqBody := EmbeddingRequest{
		Model: "text-embedding-3-small",
		Input: texts,
	}
	jsonData, _ := json.Marshal(reqBody)
	req, _ := http.NewRequest("POST", "https://api.openai.com/v1/embeddings", bytes.NewReader(jsonData))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OPENAI_API_KEY"))

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	var result EmbeddingResponse
	json.Unmarshal(body, &result)

	var vectors [][]float64
	for _, d := range result.Data {
		vectors = append(vectors, d.Embedding)
	}
	return vectors, nil
}

func hybridSearch(query string, topK int) []Doc {
	bm25Results := bm25Search(query, chunks, topK)
	vectorResults := vectorSearch(query, topK)

	// 加权融合：BM25 占 40%，向量占 60%
	merged := make(map[string]float64)
	for _, d := range bm25Results {
		merged[d.Text] += d.Score * 0.4
	}
	for _, d := range vectorResults {
		merged[d.Text] += d.Score * 0.6
	}

	var results []Doc
	for text, score := range merged {
		results = append(results, Doc{Text: text, Score: score})
	}
	sort.Slice(results, func(i, j int) bool {
		return results[i].Score > results[j].Score
	})
	if len(results) > 5 {
		results = results[:5]
	}
	return results
}

func cosineSimilarity(a, b []float64) float64 {
	dot, normA, normB := 0.0, 0.0, 0.0
	for i := range a {
		dot += a[i] * b[i]
		normA += a[i] * a[i]
		normB += b[i] * b[i]
	}
	return dot / (math.Sqrt(normA) * math.Sqrt(normB))
}

            .sorted(Map.Entry.<TextSegment, Double>comparingByValue().reversed())
            .limit(topK)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }

    // 3. 混合检索：BM25 + 向量 加权融合
    public List<TextSegment> hybridSearch(String query, int topK) {
        List<TextSegment> bm25Results = bm25Search(query, topK);
        List<TextSegment> vectorResults = vectorSearch(query, topK);

        // 加权融合：BM25 占 40%，向量占 60%
        Map<TextSegment, Double> merged = new LinkedHashMap<>();
        for (int i = 0; i < bm25Results.size(); i++) {
            merged.merge(bm25Results.get(i), (topK - i) * 0.4, Double::sum);
        }
        for (int i = 0; i < vectorResults.size(); i++) {
            merged.merge(vectorResults.get(i), (topK - i) * 0.6, Double::sum);
        }

        return merged.entrySet().stream()
            .sorted(Map.Entry.<TextSegment, Double>comparingByValue().reversed())
            .limit(5)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }
}

12.3.3 进阶：重排序（Reranker）

向量检索速度快但精度有限，返回 Top-20 中可能有噪声。重排序模型对这 20 条结果做精细打分，选出最相关的 Top-5。

📊 两阶段检索：快召回 + 精排序
  1. 召回阶段：向量检索快速从百万文档中召回 Top-50（毫秒级）
  2. 重排阶段：Cross-Encoder 模型对 50 条逐一精细打分，选 Top-5（百毫秒级）

Cross-Encoder 比 Bi-Encoder（向量检索）精度高，但速度慢 100 倍，所以只用在小规模候选集上。

func rerank(query string, candidates []string, topN int) []string {
	reqBody := RerankRequest{
		Query:     query,
		Documents: candidates,
		TopN:      topN,
	}
	jsonData, _ := json.Marshal(reqBody)
	req, _ := http.NewRequest("POST", "https://api.cohere.ai/v1/rerank", bytes.NewReader(jsonData))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+os.Getenv("COHERE_API_KEY"))

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return candidates[:topN]
	}
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	var result RerankResponse
	json.Unmarshal(body, &result)

	// 按 relevance_score 降序排序
	sort.Slice(result.Results, func(i, j int) bool {
		return result.Results[i].RelevanceScore > result.Results[j].RelevanceScore
	})

	var reranked []string
	for _, r := range result.Results {
		reranked = append(reranked, candidates[r.Index])
	}
	return reranked
}

12.4.2 HyDE（假设性文档嵌入）

有时候用户的问题太短，向量化后语义信息不足。HyDE 的思路是：先让 LLM 生成一个"假回答"，用假回答去检索，再用检索结果生成真回答。

🔄 HyDE 流程

用户问题："RAG 怎么做？" →
LLM 生成假回答："RAG 是检索增强生成技术，通过先检索相关文档再生成回答……" →
用假回答向量检索（假回答比短问题语义更丰富）→
用检索结果生成真回答

12.4.3 父子分块（Parent-Child Chunking）

检索时用小块（精确匹配），生成时用大块（上下文完整）。

策略说明 
# 大块（Parent Chunk）：512 字符 → 存储、生成时使用
# 小块（Child Chunk）：128 字符 → 检索时使用

# 流程：
# 1. 文档先分成 512 字符的大块
# 2. 每个大块再细分为 128 字符的小块
# 3. 小块做 Embedding 并存储，但保留 parent_id
# 4. 检索时匹配小块 → 找到对应的大块 → 用大块作为上下文

# 优势：检索精度高（小块精确），上下文完整（大块信息丰富）

12.7 GraphRAG：知识图谱增强检索

传统向量 RAG 有一个致命弱点：只能检索局部相似片段，无法回答全局性问题。比如"整个系统架构包含哪些模块？"这样的问题需要跨文档、跨段落的全局视角。GraphRAG 通过构建知识图谱，让 RAG 具备"关联推理"的能力。

💡 GraphRAG 的核心思想

不只是把文档切成块存向量库，而是先从文档中抽取实体和关系，构建一张知识图谱。检索时不是做向量相似度搜索，而是在图上进行子图检索和图遍历，找到与问题相关的实体及其关联信息。

12.8 Agentic RAG：Agent 驱动的自适应检索

传统 RAG 是一个固定流水线：用户提问 → 检索 → 拼接上下文 → LLM 生成。无论问题难易，流程都一样。但这不合理——有些问题不需要检索，有些需要检索多次，有些需要先检索再反思再检索。Agentic RAG 把 Agent 引入检索循环，让 Agent 动态决策是否检索、检索几次、何时停止。

💡 Agentic RAG 的核心思想

不再是一条固定的"检索→生成"流水线，而是让 Agent 作为检索编排者，根据问题难度动态决定检索策略：简单问题直接回答，复杂问题多轮检索+反思，检索不充分则换角度重搜，直到信息充分才生成最终回答。

12.8.1 传统 RAG vs Agentic RAG

🔴 传统 RAG Pipeline
  • 固定流程：Query → Retrieve → Generate
  • 每次提问都检索，不管是否需要
  • 只检索一次，不管结果够不够
  • 检索结果直接拼接，不做质量判断
  • 无法根据结果调整检索策略
🟢 Agentic RAG
  • 动态流程：Agent 自主决策每一步
  • 判断是否需要检索（简单问题跳过）
  • 支持多轮检索，直到信息充分
  • 每次检索后反思：结果是否相关？
  • 可以改写 Query 换角度重新检索

class RAGState(TypedDict):
    question: str
    retrieved_docs: List[str]
    answer: str
    reflection: str       # 反思结果
    search_count: int     # 检索次数

def decide_to_search(state: RAGState) -> str:
    """Agent 决策：是否需要检索"""
    question = state["question"]
    # 简单问题不检索（如打招呼、算术题）
    if is_simple_question(question):
        return "generate_direct"
    # 已有文档是否充分？
    if state.get("sufficient", False):
        return "generate"
    # 最多检索 3 轮，防止死循环
    if state.get("search_count", 0) >= 3:
        return "generate"
    return "search"

def search(state: RAGState) -> RAGState:
    """执行检索：Query 改写 + 混合检索 + 重排序"""
    question = state["question"]
    count = state.get("search_count", 0)
    
    # 第 2 轮起，根据反思结果改写 Query
    if count > 0:
        question = llm_rewrite_query(question, state.get("reflection", ""))
    
    docs = hybrid_search(question, top_k=10)
    ranked_docs = reranker.rerank(docs, question, top_n=5)
    
    state["retrieved_docs"] = state.get("retrieved_docs", []) + ranked_docs
    state["search_count"] = count + 1
    return state

def reflect(state: RAGState) -> RAGState:
    """反思：检索结果是否足以回答问题？"""
    prompt = f"""问题：{state['question']}
已有文档：{state['retrieved_docs']}
判断：当前文档是否足以完整回答问题？
如果不足，指出缺少什么信息。"""
    result = llm.invoke(prompt)
    state["reflection"] = result
    state["sufficient"] = "信息充分" in result
    return state

def generate(state: RAGState) -> RAGState:
    """生成最终回答"""
    context = "\n\n".join(state["retrieved_docs"])
    state["answer"] = llm.invoke(f"基于以下信息回答问题：\n{context}\n\n问题：{state['question']}")
    return state

workflow.add_node("decide", decide_to_search)
workflow.add_node("search", search)
workflow.add_node("reflect", reflect)
workflow.add_node("generate", generate)

workflow.set_entry_point("decide")
workflow.add_conditional_edges("decide", decide_to_search, {
    "search": "search",
    "generate": "generate",
    "generate_direct": "generate"
})
workflow.add_edge("search", "reflect")
workflow.add_conditional_edges("reflect", lambda s: "search" if not s["sufficient"] and s["search_count"] < 3 else "generate")
workflow.add_edge("generate", END)

  • 最大检索次数限制：通常设 3 轮，超过则用已有信息生成（宁可答案不完美，也不能死循环）
  • 反思质量评估：反思模型判断信息充分性，避免无意义的重复检索
  • Query 改写递进：每轮检索改写 Query，避免相同查询返回相同结果
  • 成本控制：每轮检索消耗 Token，Agent 需要在质量和成本间权衡
🔥 2026 面试视角：Agentic RAG vs 传统 RAG

传统 RAG是"流水线工人"——固定流程，不管实际情况。
Agentic RAG是"调研员"——先判断要不要查资料，查了一轮发现不够再查，查到够为止。

核心区别：Agentic RAG 引入了决策环节（是否检索） + 反思环节（够不够） + 迭代能力（换角度重搜），这是从"工具"升级为智能体的关键。

12.9 代码知识库：tree-sitter 符号感知分块

前面的章节讲的都是文本 RAG——把文档切成固定大小的块。但如果你要建一个代码知识库，比如把整个项目的源码放进去让 AI 回答"handle_messages 函数是干什么的"，固定大小分块会出大问题。

12.9.1 tree-sitter 是什么？

tree-sitter 是一个增量解析库，能将源码解析为 AST（抽象语法树）。它的核心能力是：给你一段代码，告诉你哪个字节范围是一个函数定义、哪个是类定义、哪个是方法。

🔧 tree-sitter 的核心能力
  • 增量解析：代码修改时只重新解析变化的部分，适合编辑器实时场景
  • 多语言支持：支持 TS/TSX/JS/Python/Rust/Go/Java 等 40+ 语言
  • AST 节点类型：function_declaration、class_declaration、method_definition 等，精确到符号级别
  • 错误恢复：代码有语法错误也能解析出有效部分，不会直接崩溃

12.10 HNSW 索引：让向量搜索从 O(n) 到 O(log n)

上一节讲了向量检索，但有一个问题被跳过了：10 万个向量，怎么搜索？

⚠️ 痛点：线性扫描的代价

最简单的做法：遍历所有向量，逐一计算余弦相似度，取 Top-K。10 万个向量就要算 10 万次余弦距离。每次计算涉及 1536 维浮点数（OpenAI text-embedding-3-small），单次搜索可能需要数百毫秒。

这在生产环境完全不可接受。

12.10.1 HNSW 原理：分层小世界图

HNSW（Hierarchical Navigable Small World） 是目前最流行的近似最近邻搜索（ANN）算法。核心思想：构建多层图结构，上层稀疏（高速公路），下层稠密（城市道路），搜索时从上层快速定位区域，再在下层精细搜索。

12.10.2 搜索过程

HNSW 搜索使用贪心最佳优先搜索 + 优先级队列：

📊 HNSW 搜索算法步骤
  1. 从顶层入口节点开始，计算当前节点与查询向量的距离
  2. 贪心搜索：检查邻居节点，如果有比当前更近的，移动过去
  3. 本层没有更近的了：下降到下一层，以上层结果为入口
  4. 到底层（Layer 0）：精细搜索，用优先级队列维护 Top-K 候选
  5. 返回 Top-K：优先级队列中相似度最高的 K 个向量

余弦距离公式：distance = 1 - cosine_similarity。距离越小越相似。

12.10.3 纯手写 vs 外部向量数据库

大多数 RAG 教程直接用 Chroma/Pinecone/Milvus，但生产中很多场景不想引入新依赖。WaLiAPI 的做法是纯 Rust 手写 HNSW，用 bincode 序列化到本地文件，零外部依赖。

方案 搜索复杂度 构建时间 内存占用 适合规模
线性扫描O(n)无需构建仅向量数据<1万

import heapq
import numpy as np

class HnswIndex:
    """简化版 HNSW 索引"""
    def __init__(self, dim: int, max_layers: int = 16):
        self.dim = dim
        self.max_layers = max_layers
        self.nodes = []          # 所有向量
        self.graphs = []         # 每层的图结构 {node_id: [neighbor_ids]}
        self.entry_point = None  # 入口节点
    
    def insert(self, vector: list):
        """插入一个向量，构建多层图"""
        node_id = len(self.nodes)
        self.nodes.append(np.array(vector))
        
        # 随机选择最高层（指数分布，越高层概率越低）
        level = int(-np.log(np.random.random()) * 1.5)
        
        # 从顶层到 level 层，每层都插入节点
        for layer in range(min(level + 1, self.max_layers)):
            if layer >= len(self.graphs):
                self.graphs.append({})
            self.graphs[layer][node_id] = []
            
            # 连接到同层的近邻节点
            if len(self.nodes) > 1:
                neighbors = self._search_layer(vector, layer, ef=10)
                self.graphs[layer][node_id] = neighbors[:5]
        
        if self.entry_point is None:
            self.entry_point = node_id
    
    def search(self, query: list, top_k: int = 5) -> list:
        """搜索 Top-K 最相似向量"""
        if self.entry_point is None:
            return []
        
        # 从顶层贪心搜索到底层
        current = self.entry_point
        for layer in range(len(self.graphs) - 1, 0, -1):
            current = self._greedy_search(query, current, layer)
        
        # 底层精细搜索
        candidates = self._search_layer(query, 0, ef=top_k * 4)
        
        # 按距离排序返回 Top-K
        distances = [(node_id, self._distance(query, self.nodes[node_id]))
                     for node_id in candidates]
        distances.sort(key=lambda x: x[1])
        return distances[:top_k]
    
    def _greedy_search(self, query, entry, layer):
        """贪心搜索：不断移动到更近的邻居"""
        current = entry
        current_dist = self._distance(query, self.nodes[current])
        while True:
            improved = False
            for neighbor in self.graphs[layer].get(current, []):
                dist = self._distance(query, self.nodes[neighbor])
                if dist < current_dist:
                    current = neighbor
                    current_dist = dist
                    improved = True
            if not improved:
                break
        return current
    
    def _search_layer(self, query, layer, ef=10):
        """在某层搜索 ef 个候选"""
        if not self.graphs[layer]:
            return []
        visited = set()
        candidates = []
        entry = self.entry_point if layer == len(self.graphs) - 1 else 0
        heapq.heappush(candidates, (self._distance(query, self.nodes[entry]), entry))
        visited.add(entry)
        results = []
        while candidates:
            dist, node = heapq.heappop(candidates)
            results.append(node)
            if len(results) >= ef:
                break
            for neighbor in self.graphs[layer].get(node, []):
                if neighbor not in visited:
                    visited.add(neighbor)
                    heapq.heappush(candidates,
                        (self._distance(query, self.nodes[neighbor]), neighbor))
        return results
    
    def _distance(self, a, b):
        """余弦距离 = 1 - 余弦相似度"""
        return 1 - np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))
    
    def save(self, path: str):
        """序列化到文件"""
        import pickle
        with open(path, 'wb') as f:\n            pickle.dump({\n                'dim': self.dim,
                'nodes': self.nodes,
                'graphs': self.graphs,
                'entry_point': self.entry_point,

    def load(cls, path: str):
        """从文件反序列化"""
        import pickle

class HnswIndex {
  private nodes: number[][] = [];
  private graphs: Map[] = [];
  private entryPoint: number | null = null;
  
  constructor(private dim: number, private maxLayers = 16) {}
  
  insert(vector: number[]): void {
    const nodeId = this.nodes.length;
    this.nodes.push(vector);
    const level = Math.floor(-Math.log(Math.random()) * 1.5);
    
    for (let layer = 0; layer <= Math.min(level, this.maxLayers - 1); layer++) {
      if (layer >= this.graphs.length) this.graphs.push(new Map());
      this.graphs[layer].set(nodeId, []);
      if (this.nodes.length > 1) {
        const neighbors = this.searchLayer(vector, layer, 10);
        this.graphs[layer].set(nodeId, neighbors.slice(0, 5));
      }
    }
    if (this.entryPoint === null) this.entryPoint = nodeId;
  }
  
  search(query: number[], topK = 5): [number, number][] {
    if (this.entryPoint === null) return [];
    let current = this.entryPoint;
    for (let layer = this.graphs.length - 1; layer > 0; layer--) {
      current = this.greedySearch(query, current, layer);
    }
    const candidates = this.searchLayer(query, 0, topK * 4);
    return candidates
      .map(id => [id, this.distance(query, this.nodes[id])] as [number, number])
      .sort((a, b) => a[1] - b[1])
      .slice(0, topK);
  }
  
  private greedySearch(query: number[], entry: number, layer: number): number {
    let current = entry;
    let currentDist = this.distance(query, this.nodes[current]);
    while (true) {
      let improved = false;
      for (const neighbor of this.graphs[layer].get(current) || []) {
        const dist = this.distance(query, this.nodes[neighbor]);
        if (dist < currentDist) {
          current = neighbor;
          currentDist = dist;
          improved = true;
        }
      }
      if (!improved) break;
    }
    return current;
  }
  
  private searchLayer(query: number[], layer: number, ef: number): number[] {
    const graph = this.graphs[layer];
    if (!graph || graph.size === 0) return [];
    const visited = new Set();
    const heap: [number, number][] = [];
    const entry = this.entryPoint!;
    heap.push([this.distance(query, this.nodes[entry]), entry]);
    visited.add(entry);
    const results: number[] = [];
    while (heap.length > 0 && results.length < ef) {
      heap.sort((a, b) => a[0] - b[0]);
      const [, node] = heap.shift()!;
      results.push(node);
      for (const neighbor of graph.get(node) || []) {
        if (!visited.has(neighbor)) {
          visited.add(neighbor);
          heap.push([this.distance(query, this.nodes[neighbor]), neighbor]);
        }
      }
    }
    return results;
  }
  
  private distance(a: number[], b: number[]): number {
    const dot = a.reduce((s, v, i) => s + v * b[i], 0);
    const norm = Math.sqrt(a.reduce((s, v) => s + v * v, 0)) *
                 Math.sqrt(b.reduce((s, v) => s + v * v, 0));
    return 1 - dot / norm;
  }
}

func (idx *HnswIndex) Insert(vector []float64) {
    nodeId := len(idx.Nodes)
    idx.Nodes = append(idx.Nodes, vector)
    level := int(-math.Log(rand.Float64()) * 1.5)
    
    for layer := 0; layer <= level; layer++ {
        if layer >= len(idx.Graphs) {
            idx.Graphs = append(idx.Graphs, make(map[int][]int))
        }
        idx.Graphs[layer][nodeId] = []int{}
        if len(idx.Nodes) > 1 {
            neighbors := idx.searchLayer(vector, layer, 10)
            if len(neighbors) > 5 {
                neighbors = neighbors[:5]
            }
            idx.Graphs[layer][nodeId] = neighbors
        }
    }
    if idx.EntryPoint == -1 {
        idx.EntryPoint = nodeId
    }
}

func (idx *HnswIndex) Search(query []float64, topK int) [][2]float64 {
    if idx.EntryPoint == -1 {
        return nil
    }
    current := idx.EntryPoint
    for layer := len(idx.Graphs) - 1; layer > 0; layer-- {
        current = idx.greedySearch(query, current, layer)
    }
    candidates := idx.searchLayer(query, 0, topK*4)
    results := make([][2]float64, 0, len(candidates))
    for _, id := range candidates {
        results = append(results, [2]float64{float64(id), idx.distance(query, idx.Nodes[id])})
    }
    sort.Slice(results, func(i, j int) bool { return results[i][1] < results[j][1] })
    if len(results) > topK {
        results = results[:topK]
    }
    return results
}

func (idx *HnswIndex) distance(a, b []float64) float64 {
    dot, normA, normB := 0.0, 0.0, 0.0
    for i := range a {
        dot += a[i] * b[i]
        normA += a[i] * a[i]
        normB += b[i] * b[i]
    }
    return 1 - dot/(math.Sqrt(normA)*math.Sqrt(normB))
}

12.11 Token 预算管理：上下文超限怎么办

RAG 检索到 20 个相关 chunk，加上对话历史，拼进 Prompt 后超过了模型的上下文窗口——这是个非常常见的生产问题。

⚠️ 痛点：Prompt 超长被拒

GPT-4o 上下文窗口 128K Token，GPT-3.5 只有 16K。如果检索结果 + 历史 + 问题总共 180K Token，API 直接报错 context_length_exceeded。

更隐蔽的问题：即使没超限，超长 Prompt 也会导致注意力稀释——模型可能忽略中间位置的关键上下文（Lost in the Middle 问题）。

12.11.1 Token 估算

不同字符的 Token 消耗不同，可以简单估算：

字符类型 估算规则 示例
ASCII（英文/代码）~4 字符 / Tokenfunction handle_messages ≈ 6 tokens
CJK（中文/日文）~2 字符 / Token处理消息函数 ≈ 3 tokens
混合内容取加权平均中英混排 ≈ 3 字符 / Token

12.11.2 三级降级策略

核心原则：context_limit = model_limit × 0.7（留 30% 给 Response）。

// 余弦相似度
func cosineSim(a, b []float32) float32 {
    dot, normA, normB := float32(0), float32(0), float32(0)
    for i := range a {
        dot += a[i] * b[i]
        normA += a[i] * a[i]
        normB += b[i] * b[i]
    }
    return dot / (float32(math.Sqrt(float64(normA))) * float32(math.Sqrt(float64(normB))))
}

    public static byte[] encodeEmbedding(float[] emb) {
        ByteBuffer buf = ByteBuffer.allocate(emb.length * 4);
        for (float v : emb) buf.putFloat(v);
        return buf.array();
    }
    
    // 向量解码
    public static float[] decodeEmbedding(byte[] blob, int dim) {
        ByteBuffer buf = ByteBuffer.wrap(blob);
        float[] emb = new float[dim];
        for (int i = 0; i < dim; i++) emb[i] = buf.getFloat();
        return emb;
    }
    
    // 余弦相似度
    public static double cosineSim(float[] a, float[] b) {
        double dot = 0, normA = 0, normB = 0;
        for (int i = 0; i < a.length; i++) {
            dot += a[i] * b[i]; normA += a[i] * a[i]; normB += b[i] * b[i];
        }
        return dot / (Math.sqrt(normA) * Math.sqrt(normB));
    }

            if (dim != queryEmb.length) continue;
            float[] emb = decodeEmbedding(rs.getBytes("embedding"), dim);
            results.add(new SearchResult(
                rs.getString("id"), rs.getString("content"),
                cosineSim(queryEmb, emb)));
        }
        results.sort((a, b) -> Double.compare(b.score, a.score));
        return results.subList(0, Math.min(topK, results.size()));

📋 八股总结 — 面试高频考点
Q1: RAG 的原理是什么？和微调有什么区别？
RAG 原理：在生成回答前，先从外部知识库检索相关文档，将检索结果作为上下文喂给 LLM 生成回答。

与微调的区别：RAG 是"外挂知识库"，不改模型参数，知识实时更新，成本低；微调是"内化知识"，修改模型参数，适合定制风格/格式，成本高。最佳实践是两者结合：微调改善输出风格，RAG 提供最新知识。
Q2: 向量数据库和传统数据库有什么区别？
传统数据库做精确匹配（WHERE id=123），向量数据库做相似度搜索（找到最接近的 K 个向量）。向量数据库使用 ANN（近似最近邻）算法索引，如 HNSW、IVF，能在百万级向量中毫秒返回结果。主流方案：Chroma（原型）、Pinecone（云托管）、Milvus（大规模自部署）、pgvector（PostgreSQL 插件）。
Q3: 什么是混合检索？为什么要做重排序？
混合检索：BM25（关键词匹配）+ 向量检索（语义匹配）加权融合。BM25 擅长精确关键词（如"GPT-4o"），向量擅长语义（如"大语言模型"）。

重排序：向量检索召回快但精度有限，用 Cross-Encoder 对候选集逐一精细打分。两阶段：先快召回 Top-50（毫秒级），再精排序 Top-5（百毫秒级），兼顾速度和精度。
Q4: RAG 系统如何评估？
从两个阶段评估：
检索质量：召回率（相关文档是否被检索到）、精确率（检索结果中有多少相关）、MRR（平均倒数排名）。
生成质量：忠实度（回答是否基于检索内容，有没有幻觉）、相关性（回答是否切题）、完整性（是否覆盖了用户问题的所有方面）。可用 Ragas 框架自动化评估。
Q5: 如何解决 RAG 检索不到相关文档的问题？
① Query 改写：用 LLM 改写用户问题，生成多个角度的查询并行检索；
② HyDE：先让 LLM 生成假回答，用假回答去检索（语义更丰富）；
③ 调整分块策略：chunk_size 太大不精确，太小语义不完整，尝试 512-1024；
④ 混合检索：加入 BM25 关键词检索，弥补向量检索关键词匹配弱的缺陷；
⑤ 增加召回量：先召回 Top-20，再用 Reranker 精选 Top-5。
Q6: RAG 检索出的内容相互冲突，Agent 该怎么取舍？
企业知识库文档版本多、来源杂，冲突是常态。三种策略：

① 元数据加权排序：根据文档的发布时间、权威等级赋予不同权重，优先采信高权威 + 最新的内容。时间维度和权威维度双重保障。

② 多智能体辩论（Multi-Agent Debate）：让不同 Agent 分别对应冲突的文档内容，相互对比论证，梳理出矛盾点，筛选出逻辑最通顺的答案。用"辩论"机制替代"单点裁决"。

③ 强制溯源：要求 Agent 输出答案必须附带检索来源，方便用户最终校验核对。把最终裁决权交给人类。

面试要点：不能只说"取最新的"，要展示元数据加权+多Agent辩论+溯源三重保障的组合思维。
Q7: 企业 RAG 怎么解决权限隔离问题？会不会泄露涉密数据？
这是企业落地 RAG 的硬性安全要求，面试高频追问。我的解决方案是RAG权限对齐：

存入阶段：在向量数据库存入数据时，给每一条向量绑定对应的访问控制权限元数据（如部门、角色、保密等级）。

检索阶段：用户发起检索请求时，系统自动带入当前用户的身份权限做过滤。

关键点：不是在"结果返回后"做过滤（可能已泄露敏感信息到上下文），而是在向量检索的源头就完成数据隔离。从根本上杜绝普通用户查到高管敏感数据的问题。

面试要点：强调"源头过滤"而非"事后过滤"，体现对数据安全闭环的理解。
Q8: 面对新闻/股价等实时更新知识库，RAG 怎么适配？
实时性是 RAG 落地的关键挑战。三种机制：

① 动态路由 Agent：先判断用户问题是否需要实时数据。需要实时 → 优先调用搜索/实时 API，不检索静态向量库；不需要 → 正常走向量库检索。类似"先思考再行动"的智能分发。

② 流式增量更新：通过消息队列监听知识库的实时变动，动态新增/更新向量数据，不用全量重构。增量更新效率高。

③ 缓存失效机制：给高频问题设置缓存时效，元数据更新后立刻清空旧缓存，保存答案的时效性，避免返回过期信息。

面试要点：三机制覆盖"判断要不要实时"（路由）、"怎么更新"（增量）、"怎么保证不过期"（缓存失效），体现全链路思维。
Q9: 怎么系统提升 RAG 问答准确率？三层体系是什么？
这是最综合的一道题，需要展示"从解析到检索到生成全链路"的完整体系思维：

第一层：深度解析层——解决文档拆分错乱
传统按字符切分会打断表格/标题/正文的关联。我用布局感知解析模型，把文档识别成标题、正文、表格、列表等完整模块，按标题层级切块，保证每一段内容语义完整不会断裂。核心思路：先理解文档结构，再按结构切。

第二层：检索增强层——多阶段精准检索
① 向量检索 + BM25 关键词检索混合模式：兼顾语义匹配和专有名词匹配。单一向量检索会丢掉精确关键词匹配能力。
② 重排序模型对初筛内容精排：这是提升准确率性价比最高的操作。初筛粗选 → 精排精选，二次提纯。
③ 扩展 Query 生成多个同义问题并行检索：解决用户提问太简短、信息不全的问题。一个问题变成多个角度同时检索。

第三层：生成校验层——实现自我纠错
答案生成前，Agent 做两次自检：
第一检：判断检索内容是否足够回答问题 → 不足就重新检索。
第二检：核对答案内容是否全部来自检索结果 → 杜绝无依据的幻觉输出。

面试要点：三层体系覆盖解析（输入端）、检索（中间端）、生成（输出端），体现全链路精度提升而非单点优化。
Q10: GraphRAG 和向量 RAG 的区别是什么？各自适合什么场景？
核心区别在检索方式：向量 RAG 做语义相似度搜索，检索的是"像"的文本块；GraphRAG 从文档中抽取实体和关系构建知识图谱，检索时做图遍历，找到的是"有关联"的信息。

向量 RAG 擅长：事实性问答（"RAG 是什么？"），单点查询，毫秒级响应，构建成本低。
GraphRAG 擅长：多跳推理（"A 公司 CEO 毕业于哪所大学？"需要跨文档追溯），关系分析（"这个故障影响哪些下游服务？"），全局摘要，跨文档关联。

GraphRAG 的代价：构建成本高（LLM 抽取实体关系消耗大量 Token），维护复杂（文档更新需同步更新图谱），有实体歧义问题（"苹果"是公司还是水果）。

最佳实践：混合使用。向量 RAG 做日常问答（90% 场景），GraphRAG 做复杂关联分析（10% 场景）。面试要点是展示"不是 GraphRAG 替代向量 RAG，而是互补"的思维。
Q11: Agentic RAG 什么时候触发检索？和传统 RAG pipeline 有什么本质区别？
本质区别：传统 RAG 是固定流水线（Query → Retrieve → Generate），每次提问都检索且只检索一次；Agentic RAG 引入 Agent 做动态决策，具备三个传统 RAG 没有的能力：

① 决策能力（是否检索）：Agent 判断问题类型——简单问题（"你好"、"1+1"）直接回答不检索，事实性问题才触发检索。触发信号包括：问题类型判断、知识边界检测（超出训练数据）、领域关键词匹配、用户显式要求。

② 反思能力（够不够）：每轮检索后，Agent 评估已有文档是否足以完整回答问题。不足则指出缺什么，触发再检索。

③ 迭代能力（换角度重搜）：根据反思结果改写 Query，换角度重新检索。比如第 1 轮搜"限流策略"，反思发现还缺"降级策略"，第 2 轮改搜"降级策略"。

防止死循环：设最大检索次数（通常 3 轮），超过则用已有信息生成。面试要点：强调"决策+反思+迭代"三重能力，这是从"工具"升级为"智能体"的关键。
Q12: 什么是 tree-sitter 符号感知分块？为什么代码 RAG 需要它？
传统按字符切分会截断函数/类，检索到的代码片段不完整——函数签名丢了、参数定义丢了，AI 根本看不懂。

tree-sitter 构建 AST，按符号边界（函数/类/方法/结构体）切分，每个 chunk 带 symbol_name/symbol_kind/line_start/line_end 元数据。检索时可以按符号类型过滤：先 hybrid_search 再 filter_by_symbol(results, "function")，只返回函数定义，过滤掉注释和 import。

支持 7 种语言：TS/TSX/JS/Python/Rust/Go/Java。面试要点：代码 RAG 和文本 RAG 的核心区别在于"分块策略"和"检索过滤"——不是按字符切，是按 AST 符号切；不是盲目返回，是按符号类型过滤。
Q13: HNSW 索引的原理是什么？为什么不直接用线性扫描？
HNSW（分层小世界图）：上层稀疏（高速公路），下层稠密（城市道路），搜索时从上层快速定位区域，再在下层精细搜索。使用贪心最佳优先搜索 + 优先级队列。

性能对比：10 万向量线性扫描 O(n) = 10 万次距离计算 ≈ 200ms；HNSW O(log n) ≈ 17 次距离计算 ≈ 2ms。快了 100 倍。

余弦距离 = 1 - cosine_similarity。纯手写 HNSW + bincode 序列化到本地文件，不依赖外部向量数据库。

面试要点：HNSW 是用空间换时间——构建图需要额外内存（每个节点存邻居指针），但搜索从 O(n) 降到 O(log n)，代价远小于收益。
Q14: RAG 系统中 Token 超限怎么办？降级策略怎么设计？
Token 估算：ASCII ~4 字符/token，CJK ~2 字符/token。context_limit = model_limit × 0.7（30% 留给 Response）。

三级降级：
Level 1 裁低分 chunk（保留高分 + 完整历史）→ Level 2 裁历史只留最近 2 条（保留高分 chunk）→ Level 3 紧急模式（无历史，top-3 chunk，简化 prompt）。

面试要点：降级不是"截断"，而是"优先级裁剪"——保高分 chunk（检索质量优先）、保最近历史（对话连贯性优先），不是简单砍掉尾部。
Q15: 如何在不引入外部向量数据库的情况下实现 RAG？
SQLite 三件套：SQLite 存数据 + BLOB 存向量（二进制序列化，比 JSON 省 75% 空间）+ 手写 HNSW 索引 + FTS5 全文索引。一个 .db 文件搞定存储 + 向量 + 全文检索，零外部依赖。

Embedding 复用：如果已有 LLM API 网关，复用其渠道调度做向量化，零额外 API Key、零额外配置。

适合中小规模知识库（万级 chunk），大规模仍需 Milvus/Qdrant。面试要点：不是"用什么向量数据库"，而是"根据场景选择合适的方案"——嵌入式方案适合桌面应用、小团队、私有部署。
Q16: MCP 如何让 AI Agent 使用知识库？
MCP（Model Context Protocol） 暴露 search/ask/build_index 等 13 个工具，任何 Agent 通过 JSON-RPC 调用。instructions 注入引导 Agent 优先用 ask/search，避免反模式（用 read_document 遍历整个文档）。

一个知识库同时服务 Claude Desktop、Cursor、Copilot 和自研 Agent——不需要为每个工具单独对接。

面试要点：MCP 解决的是"工具调用标准化"问题，就像 USB-C 之于充电器。RAG + MCP = 知识库即服务。
Q17: Deep Research 和 Agentic RAG 有什么区别？
Agentic RAG：检索 → 反思 → 再检索，侧重"信息够不够"，输出简短回答，通常 ≤ 3 轮。是"答题思维"。

Deep Research：检索 → 分析 → 追问 → 再检索 → 综合，侧重"信息全不全"，输出综合报告，通常 3-10 轮。是"调研思维"。

典型场景对比："限流策略和降级策略分别是什么？"用 Agentic RAG；"分析所有模块的技术选型和设计权衡"用 Deep Research。

面试要点：Agentic RAG 侧重宽度（够不够），Deep Research 侧重深度（全不全）。不是替代关系，是互补关系。
